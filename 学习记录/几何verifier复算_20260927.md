## 四、复算脚本（可重跑）

```python
# 依赖：仅 numpy；不需要 GPU、不需要模型权重
# 用法：cd "D:/pycode/晶圆图研究" && python 本脚本
import json, numpy as np

X = np.load("/tmp/X.npy"); y = np.load("/tmp/y.npy"); sp = np.load("/tmp/sp.npy")
rows = [json.loads(l) for l in open("data/manifest.jsonl", encoding="utf-8")]
ok = [r for r in rows if r["features"].get("status") == "ok"]
ids  = np.array([r["sample_id"] for r in ok])
lots = np.array([r["lot_name"]  for r in ok])
CLASSES = sorted(set(y))

bids = {json.loads(l)["sample_id"] for l in open("benchmark/all_requests.jsonl", encoding="utf-8")
        if json.loads(l).get("task") == "classification"}
m  = np.array([i in bids for i in ids])
tr = sp == "train"
Xtr, ytr, ltr = X[tr], y[tr], lots[tr]
Xte, yte = X[m], y[m]

def lot_strat(k, seed):
    """按 lot 分层抽 k 个（比例分配）"""
    rng = np.random.RandomState(seed)
    ul, cn = np.unique(ltr, return_counts=True)
    take = np.floor(cn / cn.sum() * k).astype(int)
    for i in np.argsort(-cn)[:k - take.sum()]:
        take[i] += 1
    idx = [rng.choice(np.where(ltr == u)[0], size=min(t, (ltr == u).sum()), replace=False)
           for u, t in zip(ul, take) if t > 0]
    return np.concatenate(idx)

def score(idx):
    mu, sd = Xtr[idx].mean(0), Xtr[idx].std(0) + 1e-9
    Ztr = (Xtr[idx] - mu) / sd
    C = {c: Ztr[ytr[idx] == c].mean(0) for c in CLASSES if (ytr[idx] == c).sum() > 0}
    Zte = (Xte - mu) / sd
    pred = np.array([min(C, key=lambda c: np.linalg.norm(C[c] - z)) for z in Zte])
    return (pred == yte).mean(), pred

for k in [50, 200, 1000]:
    accs = [score(lot_strat(k, s))[0] for s in (3407, 1, 2, 3, 4)]
    print(f"k={k:<5} {np.mean(accs):.4f}  范围 {min(accs):.4f}~{max(accs):.4f}")
a, pred = score(np.arange(len(Xtr)))
print(f"k=全部  {a:.4f}")
for c in CLASSES:
    s = yte == c
    if s.sum(): print(f"  {c:<10} {(pred[s]==c).sum():>3}/{s.sum():<3} = {(pred[s]==c).mean():.2f}")
```

**说明**：`X` / `y` / `sp` 三个数组由 `data/manifest.jsonl` 的 `features` 字段构造，31 维依次为：
`defect_ratio`(1) + `centroid_radius_r / mean_radius_r / std_radius_r / defect_extent_r / radial_span_r`(5)
+ `radial_density`(10) + `angular_probability`(12)
+ `resultant_length / angle_entropy_bits / anisotropy_min_over_max`(3)。
构造代码见本轮会话；`status != "ok"` 的 2 条被剔除，故 5904 → **5902**。

## 五、这次复算暴露的两件事

1. **口径不一致比数字错误更隐蔽。** 初稿那张表把 **test 589** 的曲线和 **基准 252** 的单点放在一起，
   两者是不同的样本集，却被当成一条连续的"标注预算曲线"来读。**先统一口径，再谈数字。**
2. **头条数字复现 ≠ 细节数字复现。** 同一个脚本，0.6693 一位不差，逐类召回却有两类差了十几个百分点。
   这说明**先前那次运行的上下文（特征子集/样本集）已经丢失** —— 一个不能重跑的测量，等于没有测量。
